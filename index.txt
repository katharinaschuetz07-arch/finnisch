    let currentCategory = 'a2';
    let currentIndex = 0;

    function updateCard() {
        const item = data[currentCategory][currentIndex];
        document.getElementById('phrase-hu').innerText = item.hu;
        document.getElementById('phrase-phonetic').innerText = item.ph;
        document.getElementById('phrase-de').innerText = item.de;
        document.getElementById('cat-label').innerText = currentCategory.toUpperCase();
        document.getElementById('counter').innerText = `${currentIndex + 1} / ${data[currentCategory].length}`;
    }

    function setCategory(cat) {
        currentCategory = cat;
        currentIndex = 0;
        updateCard();
        speak();
    }

    function nextCardWithAudio() {
        currentIndex = (currentIndex + 1) % data[currentCategory].length;
        updateCard();
        speak();
    }

    function prevCard() {
        currentIndex = (currentIndex - 1 + data[currentCategory].length) % data[currentCategory].length;
        updateCard();
        speak();
    }

    function speak() {
        const text = data[currentCategory][currentIndex].hu;
        const msg = new SpeechSynthesisUtterance(text);
        msg.lang = 'hu-HU';
        msg.rate = 0.85;
        window.speechSynthesis.speak(msg);
    }

    updateCard();
